// Original classes dominating a Witch.
// IsVictimWitch is weight 0, same as the stock victim criteria, so these rules
// tie the generic domination rule instead of replacing it.
// Scout's generic rule still carries 30PercentChance, so it still wins outright
// on that roll and these lines only play the rest of the time.
// Pyro has no Witch lines; the generic pool is unchanged.
//
// skip <seconds> starts that far into the sound (parenthetical marks).
// cutoff <seconds> stops the sound at that time (bracket marks).
// predelay 2.5 matches stock domination scenes so the death scream finishes.

criterion "IsVictimWitch" "victimclass" "Witch" "required" weight 0

response "WitchScoutDominate" predelay "2.5"
{
	speak "Scout.Domination20"
	speak "Scout.Misc02"
	speak "scout_sf12_seeking12"
	speak "scout_sf12_badmagic06"
	speak "scout_item_unicorn_domination06"
}
Rule WitchScoutDominate
{
	criteria ConceptKilledPlayer IsScout IsDominated IsVictimWitch
	ApplyContext "ScoutKillSpeech:1:10"
	ApplyContext "IsDominating:1:10"
	Response WitchScoutDominate
}
